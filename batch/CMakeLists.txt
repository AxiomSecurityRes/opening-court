cmake_minimum_required(VERSION 3.16)
project(opening_court_batch CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_executable(oc_batch src/main.cpp)
if(MSVC)
  target_compile_options(oc_batch PRIVATE /utf-8 /W3)
else()
  target_compile_options(oc_batch PRIVATE -Wall -Wextra -O2)
endif()
